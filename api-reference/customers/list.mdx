title: 'Listar Clientes'
api: 'GET https://tu-sitio.com/wp-json/latepoint/v1/customers'

---

## Descripción

Este endpoint devuelve una lista paginada de todos los clientes registrados en el sistema. Puedes filtrar, buscar y ordenar los resultados según tus necesidades.

## Autenticación

<ParamField header="X-API-Key" type="string" required>
  Tu API Key de LatePoint con permisos de lectura
</ParamField>

## Parámetros de Consulta

### Paginación

<ParamField query="page" type="integer" default="1">
  Número de página a recuperar
</ParamField>

<ParamField query="per_page" type="integer" default="20">
  Número de clientes por página (máximo 100)
</ParamField>

### Filtros

<ParamField query="status" type="string">
  Filtrar por estado del cliente
  
  **Valores posibles:**
  - `active` - Clientes activos
  - `inactive` - Clientes inactivos
  - `blocked` - Clientes bloqueados
</ParamField>

<ParamField query="created_after" type="string">
  Mostrar solo clientes creados después de esta fecha (formato: YYYY-MM-DD)
</ParamField>

<ParamField query="created_before" type="string">
  Mostrar solo clientes creados antes de esta fecha (formato: YYYY-MM-DD)
</ParamField>

<ParamField query="has_bookings" type="boolean">
  Filtrar clientes que tienen o no tienen reservas
  
  - `true` - Solo clientes con reservas
  - `false` - Solo clientes sin reservas
</ParamField>

<ParamField query="last_booking_after" type="string">
  Clientes con última reserva después de esta fecha (formato: YYYY-MM-DD)
</ParamField>

<ParamField query="last_booking_before" type="string">
  Clientes con última reserva antes de esta fecha (formato: YYYY-MM-DD)
</ParamField>

<ParamField query="agent_id" type="integer">
  Filtrar clientes que han tenido citas con un agente específico
</ParamField>

<ParamField query="service_id" type="integer">
  Filtrar clientes que han usado un servicio específico
</ParamField>

<ParamField query="location_id" type="integer">
  Filtrar clientes que han visitado una ubicación específica
</ParamField>

### Búsqueda

<ParamField query="search" type="string">
  Buscar clientes por nombre, email o teléfono
</ParamField>

<ParamField query="email" type="string">
  Buscar cliente por email exacto
</ParamField>

<ParamField query="phone" type="string">
  Buscar cliente por número de teléfono
</ParamField>

### Ordenamiento

<ParamField query="order_by" type="string" default="created_at">
  Campo por el cual ordenar los resultados
  
  **Valores posibles:**
  - `id` - Por ID del cliente
  - `first_name` - Por nombre
  - `last_name` - Por apellido
  - `email` - Por email
  - `created_at` - Por fecha de registro
  - `last_booking_date` - Por fecha de última reserva
  - `total_bookings` - Por número total de reservas
</ParamField>

<ParamField query="order_direction" type="string" default="desc">
  Dirección del ordenamiento
  
  **Valores posibles:**
  - `asc` - Ascendente
  - `desc` - Descendente
</ParamField>

### Campos Incluidos

<ParamField query="include" type="string">
  Campos adicionales a incluir en la respuesta (separados por coma)
  
  **Valores posibles:**
  - `bookings` - Incluir reservas del cliente
  - `stats` - Incluir estadísticas del cliente
  - `custom_fields` - Incluir campos personalizados
  - `notes` - Incluir notas del cliente
  - `avatar` - Incluir URL del avatar
</ParamField>

### Paginación Avanzada

<CodeGroup>

```javascript JavaScript
// Función para obtener todos los clientes con paginación
async function getAllCustomers() {
  let allCustomers = [];
  let currentPage = 1;
  let hasNextPage = true;
  
  while (hasNextPage) {
    const params = new URLSearchParams({
      page: currentPage,
      per_page: 100,
      include: 'stats'
    });
    
    const response = await fetch(`https://tu-sitio.com/wp-json/latepoint/v1/customers?${params}`, {
      headers: {
        'X-API-Key': 'lp_live_1234567890abcdef'
      }
    });
    
    const result = await response.json();
    
    if (result.status === 'success') {
      allCustomers = allCustomers.concat(result.data);
      hasNextPage = result.meta.has_next_page;
      currentPage++;
      
      console.log(`Página ${result.meta.current_page}/${result.meta.total_pages} - ${result.data.length} clientes`);
    } else {
      throw new Error('Error al obtener clientes');
    }
  }
  
  return allCustomers;
}

// Uso
const allCustomers = await getAllCustomers();
console.log(`Total de clientes obtenidos: ${allCustomers.length}`);
```

### 2. Búsqueda de Clientes en Tiempo Real

```javascript
// Función de búsqueda con debounce para autocompletado
function createCustomerSearch() {
  let searchTimeout;
  
  return function searchCustomers(query, callback) {
    clearTimeout(searchTimeout);
    
    searchTimeout = setTimeout(async () => {
      if (query.length < 2) {
        callback([]);
        return;
      }
      
      const params = new URLSearchParams({
        search: query,
        status: 'active',
        per_page: 10,
        order_by: 'last_name'
      });
      
      try {
        const response = await fetch(`/wp-json/latepoint/v1/customers?${params}`, {
          headers: { 'X-API-Key': 'tu_api_key' }
        });
        
        const result = await response.json();
        
        if (result.status === 'success') {
          const customers = result.data.map(customer => ({
            id: customer.id,
            name: `${customer.first_name} ${customer.last_name}`,
            email: customer.email,
            phone: customer.phone
          }));
          
          callback(customers);
        } else {
          callback([]);
        }
      } catch (error) {
        console.error('Error en búsqueda:', error);
        callback([]);
      }
    }, 300); // Debounce de 300ms
  };
}

// Uso
const searchCustomers = createCustomerSearch();
searchCustomers('juan', (customers) => {
  console.log('Resultados:', customers);
});
```

### 3. Análisis de Clientes Inactivos

```javascript
// Identificar clientes que necesitan seguimiento
async function getInactiveCustomersAnalysis() {
  const sixMonthsAgo = new Date();
  sixMonthsAgo.setMonth(sixMonthsAgo.getMonth() - 6);
  const dateFilter = sixMonthsAgo.toISOString().split('T')[0];
  
  const params = new URLSearchParams({
    status: 'active',
    has_bookings: 'true',
    last_booking_before: dateFilter,
    include: 'stats',
    order_by: 'last_booking_date',
    order_direction: 'asc',
    per_page: 100
  });
  
  const response = await fetch(`/wp-json/latepoint/v1/customers?${params}`, {
    headers: { 'X-API-Key': 'tu_api_key' }
  });
  
  const result = await response.json();
  
  if (result.status === 'success') {
    return result.data.map(customer => {
      const daysSinceLastBooking = Math.floor(
        (new Date() - new Date(customer.stats.last_booking_date)) / (1000 * 60 * 60 * 24)
      );
      
      return {
        id: customer.id,
        name: `${customer.first_name} ${customer.last_name}`,
        email: customer.email,
        lastBookingDate: customer.stats.last_booking_date,
        daysSinceLastBooking,
        totalBookings: customer.stats.total_bookings,
        totalSpent: customer.stats.total_spent,
        riskLevel: daysSinceLastBooking > 365 ? 'high' : 
                  daysSinceLastBooking > 180 ? 'medium' : 'low'
      };
    });
  }
  
  return [];
}
```

### 4. Exportar Lista de Clientes

```javascript
// Exportar todos los clientes a CSV
async function exportCustomersToCSV() {
  let allCustomers = [];
  let currentPage = 1;
  let hasNextPage = true;
  
  // Obtener todos los clientes
  while (hasNextPage) {
    const params = new URLSearchParams({
      page: currentPage,
      per_page: 100,
      include: 'stats,custom_fields',
      status: 'active'
    });
    
    const response = await fetch(`/wp-json/latepoint/v1/customers?${params}`, {
      headers: { 'X-API-Key': 'tu_api_key' }
    });
    
    const result = await response.json();
    
    if (result.status === 'success') {
      allCustomers = allCustomers.concat(result.data);
      hasNextPage = result.meta.has_next_page;
      currentPage++;
    } else {
      break;
    }
  }
  
  // Convertir a CSV
  const csvHeaders = [
    'ID', 'Nombre', 'Apellido', 'Email', 'Teléfono', 
    'Estado', 'Fecha Registro', 'Total Reservas', 'Total Gastado', 'Última Reserva'
  ];
  
  const csvRows = allCustomers.map(customer => [
    customer.id,
    customer.first_name,
    customer.last_name,
    customer.email,
    customer.phone,
    customer.status,
    customer.created_at,
    customer.stats?.total_bookings || 0,
    customer.stats?.total_spent || '0.00',
    customer.stats?.last_booking_date || 'N/A'
  ]);
  
  const csvContent = [csvHeaders, ...csvRows]
    .map(row => row.map(field => `"${field}"`).join(','))
    .join('\n');
  
  // Descargar archivo
  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = `clientes_${new Date().toISOString().split('T')[0]}.csv`;
  link.click();
  
  return allCustomers.length;
}
```

### 3. Caché de Resultados

```javascript
// Implementar caché simple para consultas frecuentes
const customerCache = new Map();

async function getCachedCustomers(cacheKey, params) {
  if (customerCache.has(cacheKey)) {
    const cached = customerCache.get(cacheKey);
    if (Date.now() - cached.timestamp < 300000) { // 5 minutos
      return cached.data;
    }
  }
  
  const response = await fetch(`/wp-json/latepoint/v1/customers?${params}`);
  const result = await response.json();
  
  customerCache.set(cacheKey, {
    data: result,
    timestamp: Date.now()
  });
  
  return result;
}
```

## Notas Importantes

<Note>
  **Rendimiento**: Para listas grandes de clientes, usa paginación y filtros apropiados para mantener un buen rendimiento.
</Note>

<Tip>
  **Campos Incluidos**: Usa el parámetro `include` solo cuando necesites información adicional para evitar respuestas innecesariamente grandes.
</Tip>

<Warning>
  **Rate Limiting**: Ten en cuenta los límites de velocidad al hacer múltiples solicitudes, especialmente al paginar a través de grandes conjuntos de datos.
</Warning>

<Info>
  **Búsqueda**: La búsqueda por texto es sensible a acentos y mayúsculas. Para mejores resultados, normaliza las consultas de búsqueda.
</Info>
