## Descripción

Este endpoint permite crear un nuevo cliente en el sistema LatePoint. Puedes proporcionar información básica o completa del cliente, incluyendo datos de contacto, dirección, preferencias y campos personalizados.

## Autenticación

<ParamField header="X-API-Key" type="string" required>
  Tu API Key de LatePoint con permisos de escritura
</ParamField>

## Cuerpo de la Solicitud

### Campos Requeridos

<ParamField body="first_name" type="string" required>
  Nombre del cliente
  
  **Validaciones:**
  - Mínimo 2 caracteres
  - Máximo 50 caracteres
  - Solo letras, espacios y guiones
</ParamField>

<ParamField body="last_name" type="string" required>
  Apellido del cliente
  
  **Validaciones:**
  - Mínimo 2 caracteres
  - Máximo 50 caracteres
  - Solo letras, espacios y guiones
</ParamField>

<ParamField body="email" type="string" required>
  Email del cliente (debe ser único)
  
  **Validaciones:**
  - Formato de email válido
  - Único en el sistema
  - Máximo 100 caracteres
</ParamField>

### Campos Opcionales

<ParamField body="phone" type="string">
  Número de teléfono del cliente
  
  **Validaciones:**
  - Formato de teléfono válido
  - Puede incluir código de país
  - Máximo 20 caracteres
</ParamField>

<ParamField body="status" type="string" default="active">
  Estado inicial del cliente
  
  **Valores posibles:**
  - `active` - Cliente activo (por defecto)
  - `inactive` - Cliente inactivo
  - `pending` - Pendiente de verificación
</ParamField>

<ParamField body="date_of_birth" type="string">
  Fecha de nacimiento del cliente (formato: YYYY-MM-DD)
</ParamField>

<ParamField body="gender" type="string">
  Género del cliente
  
  **Valores posibles:**
  - `male` - Masculino
  - `female` - Femenino
  - `other` - Otro
  - `prefer_not_to_say` - Prefiere no decir
</ParamField>

```javascript JavaScript
// Función para crear cliente desde formulario
async function createCustomerFromForm(formData) {
  try {
    // Validar datos del formulario
    const validation = validateCustomerData(formData);
    if (!validation.isValid) {
      throw new Error('Datos inválidos: ' + validation.errors.join(', '));
    }
    
    // Preparar datos del cliente
    const customerData = {
      first_name: formData.firstName.trim(),
      last_name: formData.lastName.trim(),
      email: formData.email.toLowerCase().trim(),
      phone: formData.phone?.trim(),
      date_of_birth: formData.dateOfBirth,
      gender: formData.gender,
      preferences: {
        language: formData.language || 'es',
        communication_preferences: {
          email_notifications: formData.emailNotifications !== false,
          sms_notifications: formData.smsNotifications === true,
          marketing_emails: formData.marketingEmails === true
        }
      },
      send_welcome_email: formData.sendWelcomeEmail !== false,
      create_wp_user: formData.createAccount === true
    };
    
    // Agregar dirección si está completa
    if (formData.street && formData.city) {
      customerData.address = {
        street: formData.street,
        city: formData.city,
        state: formData.state,
        postal_code: formData.postalCode,
        country: formData.country || 'MX'
      };
    }
    
    // Agregar campos personalizados
    if (formData.customFields) {
      customerData.custom_fields = formData.customFields;
    }
    
    // Crear cliente
    const response = await fetch('/wp-json/latepoint/v1/customers', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-API-Key': 'tu_api_key'
      },
      body: JSON.stringify(customerData)
    });
    
    if (!response.ok) {
      throw new Error(`Error HTTP: ${response.status}`);
    }
    
    const result = await response.json();
    
    if (result.status === 'success') {
      return {
        success: true,
        customer: result.data,
        message: 'Cliente creado exitosamente'
      };
    } else {
      throw new Error(result.error.message);
    }
  } catch (error) {
    console.error('Error al crear cliente:', error);
    return {
      success: false,
      error: error.message
    };
  }
}

function validateCustomerData(data) {
  const errors = [];
  
  if (!data.firstName || data.firstName.trim().length < 2) {
    errors.push('El nombre debe tener al menos 2 caracteres');
  }
  
  if (!data.lastName || data.lastName.trim().length < 2) {
    errors.push('El apellido debe tener al menos 2 caracteres');
  }
  
  if (!data.email || !isValidEmail(data.email)) {
    errors.push('Email inválido');
  }
  
  if (data.phone && !isValidPhone(data.phone)) {
    errors.push('Teléfono inválido');
  }
  
  return {
    isValid: errors.length === 0,
    errors
  };
}

function isValidEmail(email) {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return emailRegex.test(email);
}

function isValidPhone(phone) {
  const phoneRegex = /^[\+]?[1-9][\d\s\-\(\)]{7,15}$/;
  return phoneRegex.test(phone);
}
```

## Validaciones Automáticas

### Validaciones de Datos

1. **Nombre y Apellido**:
   - Mínimo 2 caracteres, máximo 50
   - Solo letras, espacios, guiones y apostrofes
   - Se eliminan espacios extra automáticamente

2. **Email**:
   - Formato válido de email
   - Único en el sistema
   - Se convierte a minúsculas automáticamente
   - Máximo 100 caracteres

3. **Teléfono**:
   - Formato válido (puede incluir código de país)
   - Se normalizan los espacios y caracteres especiales
   - Máximo 20 caracteres

4. **Fecha de Nacimiento**:
   - Formato YYYY-MM-DD
   - No puede ser fecha futura
   - Edad mínima configurable (por defecto 13 años)

### Validaciones de Negocio

1. **Referencias**:
   - Los IDs de agente, ubicación y servicio deben existir y estar activos
   - Las preferencias de horario deben ser válidas

2. **Campos Personalizados**:
   - Deben coincidir con la configuración definida
   - Validaciones específicas según el tipo de campo

3. **Dirección**:
   - Código de país debe ser válido (ISO 3166-1 alpha-2)
   - Código postal debe coincidir con el formato del país

```javascript
    const response = await fetch('/wp-json/latepoint/v1/customers', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-API-Key': 'tu_api_key'
      },
      body: JSON.stringify(customerData)
    });
    
    const result = await response.json();
    
    if (result.status === 'success') {
      return result.data;
    } else {
      throw new Error(result.error.message);
    }
  } catch (error) {
    console.error('Error al crear cliente:', error);
    throw error;
  }
}
```

```javascript
// Importar múltiples clientes
async function importCustomers(customersData) {
  const results = {
    success: [],
    errors: []
  };
  
  for (const customerData of customersData) {
    try {
      // Verificar si el email ya existe
      const emailExists = await checkEmailAvailability(customerData.email);
      if (!emailExists) {
        results.errors.push({
          email: customerData.email,
          error: 'Email ya existe'
        });
        continue;
      }
      
      // Crear cliente
      const customer = await createCustomer({
        ...customerData,
        send_welcome_email: false, // No enviar emails masivos
        create_wp_user: false // No crear usuarios masivamente
      });
      
      results.success.push(customer);
      
      // Pausa para evitar rate limiting
      await new Promise(resolve => setTimeout(resolve, 100));
      
    } catch (error) {
      results.errors.push({
        email: customerData.email,
        error: error.message
      });
    }
  }
  
  return results;
}
```

```javascript
// Validar datos antes de enviar
function validateCustomerForm(formData) {
  const errors = {};
  
  // Validar nombre
  if (!formData.first_name || formData.first_name.trim().length < 2) {
    errors.first_name = 'El nombre debe tener al menos 2 caracteres';
  }
  
  // Validar email
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!formData.email || !emailRegex.test(formData.email)) {
    errors.email = 'Email inválido';
  }
  
  // Validar teléfono (opcional)
  if (formData.phone) {
    const phoneRegex = /^[\+]?[1-9][\d\s\-\(\)]{7,15}$/;
    if (!phoneRegex.test(formData.phone)) {
      errors.phone = 'Formato de teléfono inválido';
    }
  }
  
  return {
    isValid: Object.keys(errors).length === 0,
    errors
  };
}
```

### 2. Manejo de Errores Robusto

```javascript
// Función con reintentos automáticos
async function createCustomerWithRetry(customerData, maxRetries = 3) {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await createCustomer(customerData);
    } catch (error) {
      if (attempt === maxRetries) {
        throw error;
      }
      
      // Solo reintentar en errores de red, no en errores de validación
      if (error.message.includes('validation') || error.message.includes('already_exists')) {
        throw error;
      }
      
      // Esperar antes del siguiente intento
      await new Promise(resolve => setTimeout(resolve, 1000 * attempt));
    }
  }
}
```

### 3. Optimización de Performance

```javascript
// Crear múltiples clientes de forma eficiente
async function createCustomersBatch(customersData, batchSize = 5) {
  const results = [];
  
  for (let i = 0; i < customersData.length; i += batchSize) {
    const batch = customersData.slice(i, i + batchSize);
    
    const batchPromises = batch.map(customerData => 
      createCustomer(customerData).catch(error => ({ error: error.message, data: customerData }))
    );
    
    const batchResults = await Promise.all(batchPromises);
    results.push(...batchResults);
    
    // Pausa entre lotes para respetar rate limits
    if (i + batchSize < customersData.length) {
      await new Promise(resolve => setTimeout(resolve, 500));
    }
  }
  
  return results;
}
```
