## Descripción

Este endpoint devuelve toda la información disponible de un cliente específico, incluyendo sus datos personales, estadísticas, reservas y campos personalizados.

## Autenticación

<ParamField header="X-API-Key" type="string" required>
  Tu API Key de LatePoint con permisos de lectura
</ParamField>

## Parámetros de Ruta

<ParamField path="id" type="integer" required>
  ID único del cliente a obtener
</ParamField>

## Parámetros de Consulta

### Campos Incluidos

<ParamField query="include" type="string">
  Campos adicionales a incluir en la respuesta (separados por coma)
  
  **Valores posibles:**
  - `bookings` - Incluir todas las reservas del cliente
  - `upcoming_bookings` - Solo reservas futuras
  - `past_bookings` - Solo reservas pasadas
  - `stats` - Incluir estadísticas detalladas
  - `custom_fields` - Incluir campos personalizados
  - `notes` - Incluir notas del cliente
  - `avatar` - Incluir información del avatar
  - `preferences` - Incluir preferencias del cliente
  - `payment_methods` - Incluir métodos de pago guardados
</ParamField>

<ParamField query="bookings_limit" type="integer" default="10">
  Límite de reservas a incluir (cuando se incluyen reservas)
</ParamField>

<ParamField query="bookings_status" type="string">
  Filtrar reservas por estado específico
  
  **Valores posibles:**
  - `pending` - Pendientes
  - `confirmed` - Confirmadas
  - `completed` - Completadas
  - `cancelled` - Canceladas
  - `no_show` - No se presentó
</ParamField>

### 1. Perfil Completo del Cliente

```javascript
// Función para obtener perfil completo del cliente
async function getCustomerProfile(customerId) {
  try {
    const params = new URLSearchParams({
      include: 'stats,upcoming_bookings,custom_fields,preferences,notes',
      bookings_limit: 10
    });
    
    const response = await fetch(
      `/wp-json/latepoint/v1/customers/${customerId}?${params}`,
      {
        headers: { 'X-API-Key': 'tu_api_key' }
      }
    );
    
    if (!response.ok) {
      throw new Error(`Error ${response.status}: ${response.statusText}`);
    }
    
    const result = await response.json();
    
    if (result.status === 'success') {
      const customer = result.data;
      
      return {
        basicInfo: {
          id: customer.id,
          name: `${customer.first_name} ${customer.last_name}`,
          email: customer.email,
          phone: customer.phone,
          status: customer.status
        },
        stats: customer.stats,
        upcomingBookings: customer.bookings || [],
        preferences: customer.preferences,
        customFields: customer.custom_fields,
        notes: customer.notes
      };
    } else {
      throw new Error(result.error.message);
    }
  } catch (error) {
    console.error('Error al obtener perfil del cliente:', error);
    throw error;
  }
}

// Uso
```

```javascript
// Función para analizar el historial de un cliente
async function analyzeCustomerHistory(customerId) {
  const params = new URLSearchParams({
    include: 'stats,bookings',
    bookings_limit: 50
  });
  
  const response = await fetch(
    `/wp-json/latepoint/v1/customers/${customerId}?${params}`,
    {
      headers: { 'X-API-Key': 'tu_api_key' }
    }
  );
  
  const result = await response.json();
  
  if (result.status === 'success') {
    const customer = result.data;
    const stats = customer.stats;
    const bookings = customer.bookings || [];
    
    // Análisis de patrones
    const analysis = {
      customerInfo: {
        name: `${customer.first_name} ${customer.last_name}`,
        memberSince: customer.created_at,
        status: customer.status
      },
      loyaltyMetrics: {
        totalBookings: stats.total_bookings,
        completionRate: ((stats.completed_bookings / stats.total_bookings) * 100).toFixed(1),
        noShowRate: ((stats.no_show_bookings / stats.total_bookings) * 100).toFixed(1),
        averageValue: stats.average_booking_value,
        totalSpent: stats.total_spent
      },
      preferences: {
        favoriteAgent: stats.favorite_agent?.display_name,
        favoriteService: stats.favorite_service?.name,
        favoriteLocation: stats.favorite_location?.name,
        bookingFrequency: stats.booking_frequency
      },
      recentActivity: {
        lastBooking: stats.last_booking_date,
        nextBooking: stats.next_booking_date,
        daysSinceLastBooking: stats.last_booking_date ? 
          Math.floor((new Date() - new Date(stats.last_booking_date)) / (1000 * 60 * 60 * 24)) : null
      },
      riskAssessment: {
        riskLevel: calculateRiskLevel(stats),
        recommendations: generateRecommendations(stats, bookings)
      }
    };
    
    return analysis;
  }
  
  throw new Error('Error al obtener datos del cliente');
}

function calculateRiskLevel(stats) {
  const daysSinceLastBooking = stats.last_booking_date ? 
    Math.floor((new Date() - new Date(stats.last_booking_date)) / (1000 * 60 * 60 * 24)) : 999;
  
  const noShowRate = (stats.no_show_bookings / stats.total_bookings) * 100;
  
  if (daysSinceLastBooking > 180 || noShowRate > 20) return 'high';
  if (daysSinceLastBooking > 90 || noShowRate > 10) return 'medium';
  return 'low';
}

function generateRecommendations(stats, bookings) {
  const recommendations = [];
  
  const daysSinceLastBooking = stats.last_booking_date ? 
    Math.floor((new Date() - new Date(stats.last_booking_date)) / (1000 * 60 * 60 * 24)) : 999;
  
  if (daysSinceLastBooking > 90) {
    recommendations.push('Enviar campaña de reactivación');
  }
  
  if (stats.no_show_bookings > 2) {
    recommendations.push('Implementar recordatorios adicionales');
  }
  
  if (stats.total_bookings > 10 && stats.total_spent > 500) {
    recommendations.push('Candidato para programa de lealtad');
  }
  
  return recommendations;
}
```

### 4. Validación de Cliente para Reservas

```javascript
// Validar cliente antes de crear una reserva
async function validateCustomerForBooking(customerId) {
  try {
    const params = new URLSearchParams({
      include: 'stats,upcoming_bookings'
    });
    
    const response = await fetch(
      `/wp-json/latepoint/v1/customers/${customerId}?${params}`,
      {
        headers: { 'X-API-Key': 'tu_api_key' }
      }
    );
    
    const result = await response.json();
    
    if (result.status === 'success') {
      const customer = result.data;
      const validation = {
        isValid: true,
        warnings: [],
        errors: [],
        customer: customer
      };
      
      // Verificar estado del cliente
      if (customer.status === 'blocked') {
        validation.isValid = false;
        validation.errors.push('Cliente bloqueado - no puede hacer reservas');
      }
      
      if (customer.status === 'inactive') {
        validation.warnings.push('Cliente inactivo - verificar información');
      }
      
      // Verificar historial de no-shows
      const noShowRate = (customer.stats.no_show_bookings / customer.stats.total_bookings) * 100;
      if (noShowRate > 20) {
        validation.warnings.push(`Alto índice de no-shows (${noShowRate.toFixed(1)}%)`);
      }
      
      // Verificar reservas pendientes
      const pendingBookings = customer.bookings?.filter(b => b.status === 'pending') || [];
      if (pendingBookings.length > 2) {
        validation.warnings.push(`Tiene ${pendingBookings.length} reservas pendientes`);
      }
      
      // Verificar información de contacto
      if (!customer.phone || !customer.email) {
        validation.warnings.push('Información de contacto incompleta');
      }
      
      return validation;
    } else {
      return {
        isValid: false,
        errors: ['Cliente no encontrado'],
        warnings: [],
        customer: null
      };
    }
  } catch (error) {
    return {
      isValid: false,
      errors: ['Error al validar cliente'],
      warnings: [],
      customer: null
    };
  }
}

// Uso
const validation = await validateCustomerForBooking(45);

if (validation.isValid) {
  if (validation.warnings.length > 0) {
    console.warn('Advertencias:', validation.warnings);
  }
  console.log('Cliente válido para reserva');
} else {
  console.error('Errores:', validation.errors);
}
```

### 2. Caché de Datos del Cliente

```javascript
// Implementar caché para datos del cliente
const customerCache = new Map();

async function getCachedCustomer(customerId, include = '') {
  const cacheKey = `${customerId}_${include}`;
  
  if (customerCache.has(cacheKey)) {
    const cached = customerCache.get(cacheKey);
    if (Date.now() - cached.timestamp < 300000) { // 5 minutos
      return cached.data;
    }
  }
  
  const params = include ? `?include=${include}` : '';
  const response = await fetch(`/wp-json/latepoint/v1/customers/${customerId}${params}`);
  const result = await response.json();
  
  customerCache.set(cacheKey, {
    data: result,
    timestamp: Date.now()
  });
  
  return result;
}
```

### 3. Manejo de Errores Robusto

```javascript
// Función con manejo completo de errores
async function safeGetCustomer(customerId, include = '') {
  try {
    const params = include ? new URLSearchParams({ include }) : '';
    const url = `/wp-json/latepoint/v1/customers/${customerId}${params ? '?' + params : ''}`;
    
    const response = await fetch(url, {
      headers: { 'X-API-Key': 'tu_api_key' }
    });
    
    if (!response.ok) {
      if (response.status === 404) {
        throw new Error('Cliente no encontrado');
      } else if (response.status === 403) {
        throw new Error('Sin permisos para acceder a este cliente');
      } else {
        throw new Error(`Error del servidor: ${response.status}`);
      }
    }
    
    const result = await response.json();
    
    if (result.status === 'success') {
      return result.data;
    } else {
      throw new Error(result.error.message);
    }
  } catch (error) {
    console.error(`Error al obtener cliente ${customerId}:`, error.message);
    throw error;
  }
}
```
